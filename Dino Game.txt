#define BUTTON_PIN 4
#define BUZZER_PIN 5

};

// --- Game States ---
enum GameState {
  STATE_MENU,
  STATE_COUNTDOWN,
  STATE_PLAYING,
  STATE_GAME_OVER
};

GameState currentState = STATE_MENU;

// --- Global Variables ---
float gameSpeed = 2.0;
unsigned long score = 0;
unsigned long highScore = 0;
unsigned long lastScoreTime = 0;

unsigned long lastFrameTime = 0;
const int FRAME_RATE = 60; // Target 60 FPS
const int FRAME_DELAY = 1000 / FRAME_RATE; // ~16ms

// Button debouncing state
int buttonState = HIGH;
int lastButtonState = HIGH;
unsigned long lastDebounceTime = 0;
unsigned long debounceDelay = 50;

// --- Entities Structs ---
struct Dinosaur {
  float y;
  float velocityY;
  bool isJumping;
  int width = 16;
  int height = 16;
  int x = 10;
  int animationFrame = 0;
  unsigned long lastAnimTime = 0;
} dino;

struct Obstacle {
  float x;
  int y;
  int width;
  int height;
  int type; // 0: small, 1: large
  bool active;
};
#define MAX_OBSTACLES 3
Obstacle obstacles[MAX_OBSTACLES];

struct Cloud {
  float x;
  int y;
  bool active;
};
#define MAX_CLOUDS 2
Cloud clouds[MAX_CLOUDS];

// --- Function Prototypes ---
void resetGame();
void showStartMenu();
void updateGame();
void drawGame(bool updateDisplay = true);
void checkCollision();
void showGameOver();
bool checkButton();
void runCountdown();

// --- Setup ---
void setup() {
  Serial.begin(115200);
  
  pinMode(BUTTON_PIN, INPUT_PULLUP);
  pinMode(BUZZER_PIN, OUTPUT);

  // Initialize OLED (Address 0x3C is standard for 128x64)
  if(!display.begin(0x3C, true)) { 
    Serial.println(F("SH1106 allocation failed"));
    for(;;); // Loop forever if display initialization fails
  }

  // Load high score from non-volatile storage
  preferences.begin("dinogame", false);
  highScore = preferences.getUInt("highscore", 0);

  // Seed random generator with unconnected analog pin
  randomSeed(analogRead(34));

  display.clearDisplay();
  display.display();
}

// --- Main Loop ---
void loop() {
  // Cap the game at Target FPS for consistent physics
  if (millis() - lastFrameTime >= FRAME_DELAY) {
    lastFrameTime = millis();
    
    // Check button every frame
    bool btnPressed = checkButton();

    switch(currentState) {
      case STATE_MENU:
        if (btnPressed) {
          currentState = STATE_COUNTDOWN;
        }
        showStartMenu();
        break;
      
      case STATE_COUNTDOWN:
        runCountdown();
        break;

      case STATE_PLAYING:
        if (btnPressed && !dino.isJumping) {
          dino.velocityY = -2.8; // Initial jump strength
          dino.isJumping = true;
          tone(BUZZER_PIN, 600, 100);
        }
        updateGame();
        drawGame();
        break;

      case STATE_GAME_OVER:
        if (btnPressed) {
          currentState = STATE_PLAYING;
          resetGame();
        }
        showGameOver();
        break;
    }
  }
}

// --- Functions ---
bool checkButton() {
  int reading = digitalRead(BUTTON_PIN);
  bool pressed = false;
  
  if (reading != lastButtonState) {
    lastDebounceTime = millis();
  }
  
  if ((millis() - lastDebounceTime) > debounceDelay) {
    if (reading != buttonState) {
      buttonState = reading;
      if (buttonState == LOW) { // LOW implies button pressed (PULLUP)
        pressed = true;
      }
    }
  }
  
  lastButtonState = reading;
  return pressed;
}

unsigned long countdownStartTime = 0;
int countdownValue = 3;

void runCountdown() {
  if (countdownStartTime == 0) {
    countdownStartTime = millis();
    countdownValue = 3;
    tone(BUZZER_PIN, 400, 100);
  }
  
  unsigned long elapsed = millis() - countdownStartTime;
  
  // Transition countdown every second
  if (elapsed > 1000 * (4 - countdownValue)) {
    countdownValue--;
    if (countdownValue > 0) {
      tone(BUZZER_PIN, 400, 100);
    } else if (countdownValue == 0) {
      tone(BUZZER_PIN, 800, 200);
    }
  }

  display.clearDisplay();
  display.setTextSize(2);
  display.setTextColor(WHITE);
  
  if (countdownValue > 0) {
    display.setCursor(SCREEN_WIDTH/2 - 6, SCREEN_HEIGHT/2 - 8);
    display.print(countdownValue);
  } else {
    display.setCursor(SCREEN_WIDTH/2 - 18, SCREEN_HEIGHT/2 - 8);
    display.print("GO!");
  }
  
  display.display();

  if (countdownValue < 0) {
    countdownStartTime = 0; // Reset for next time
    resetGame();
    currentState = STATE_PLAYING;
  }
}

void resetGame() {
  score = 0;
  gameSpeed = 2.0;
  lastScoreTime = millis();
  
  dino.y = 52 - dino.height;
  dino.velocityY = 0;
  dino.isJumping = false;
  
  for (int i = 0; i < MAX_OBSTACLES; i++) {
    obstacles[i].active = false;
  }
  for (int i = 0; i < MAX_CLOUDS; i++) {
    clouds[i].active = false;
  }
}

void updateGame() {
  unsigned long currentTime = millis();
  
  // Update Score & Speed every 100ms
  if (currentTime - lastScoreTime > 100) {
    score++;
    lastScoreTime = currentTime;
    
    // Increase difficulty speed every 100 points
    if (score % 100 == 0) {
      gameSpeed += 0.2;
      tone(BUZZER_PIN, 1000, 150); // Level up notification sound
    }
  }

  // Update Dinosaur Physics
  if (dino.isJumping) {
    dino.y += dino.velocityY;
    dino.velocityY += 0.15; // Apply Gravity
    
    // Check if landed on the ground line (y = 52)
    if (dino.y >= (52 - dino.height)) {
      dino.y = 52 - dino.height;
      dino.isJumping = false;
      dino.velocityY = 0;
    }
  } else {
    // Ground Animation frame update
    if (currentTime - dino.lastAnimTime > (200 / gameSpeed)) { 
      dino.animationFrame = (dino.animationFrame + 1) % 2;
      dino.lastAnimTime = currentTime;
    }
  }

  // Update Obstacles Movement
  for (int i = 0; i < MAX_OBSTACLES; i++) {
    if (obstacles[i].active) {
      obstacles[i].x -= gameSpeed;
      if (obstacles[i].x + obstacles[i].width < 0) {
        obstacles[i].active = false; // Despawn when out of screen bounds
      }
    }
  }

  // Spawn new Obstacles randomly
  bool canSpawn = true;
  // Ensure we don't spawn obstacles too close to each other
  for (int i = 0; i < MAX_OBSTACLES; i++) {
    if (obstacles[i].active && obstacles[i].x > SCREEN_WIDTH - 50 - random(0, 40)) {
      canSpawn = false;
    }
  }
  
  if (canSpawn && random(100) < 5) {
    for (int i = 0; i < MAX_OBSTACLES; i++) {
      if (!obstacles[i].active) {
        obstacles[i].active = true;
        obstacles[i].type = random(2);
        obstacles[i].width = (obstacles[i].type == 0) ? 8 : 16;
        obstacles[i].height = 16;
        obstacles[i].x = SCREEN_WIDTH;
        obstacles[i].y = 52 - obstacles[i].height; // Rest on the ground
        break;
      }
    }
  }

  // Update Clouds Movement
  for (int i = 0; i < MAX_CLOUDS; i++) {
    if (clouds[i].active) {
      clouds[i].x -= (gameSpeed * 0.3); // Clouds have a slower parallax effect
      if (clouds[i].x + 16 < 0) {
        clouds[i].active = false;
      }
    }
  }

  // Spawn Clouds Randomly
  if (random(100) < 2) {
    for (int i = 0; i < MAX_CLOUDS; i++) {
      if (!clouds[i].active) {
        clouds[i].active = true;
        clouds[i].x = SCREEN_WIDTH;
        clouds[i].y = random(5, 20); // Random height for sky
        break;
      }
    }
  }

  checkCollision();
}

void checkCollision() {
  // Dino Hitbox (shrunk slightly for fair gameplay against sharp corners)
  int dX = dino.x + 3;
  int dY = dino.y + 3;
  int dW = dino.width - 6;
  int dH = dino.height - 4;

  for (int i = 0; i < MAX_OBSTACLES; i++) {
    if (obstacles[i].active) {
      int oX = obstacles[i].x + 2;
      int oY = obstacles[i].y + 2;
      int oW = obstacles[i].width - 4;
      int oH = obstacles[i].height - 4;

      // AABB Collision Detection
      if (dX < oX + oW && dX + dW > oX &&
          dY < oY + oH && dY + dH > oY) {
        
        tone(BUZZER_PIN, 150, 400); // Game over crash sound
        currentState = STATE_GAME_OVER;
        
        // Save High Score persistently to flash memory
        if (score > highScore) {
          highScore = score;
          preferences.putUInt("highscore", highScore);
        }
      }
    }
  }
}

void drawGame(bool updateDisplay) {
  display.clearDisplay();

  // Draw Ground Line
  display.drawLine(0, 52, SCREEN_WIDTH, 52, WHITE);

  // Draw Clouds
  for (int i = 0; i < MAX_CLOUDS; i++) {
    if (clouds[i].active) {
      display.drawBitmap(clouds[i].x, clouds[i].y, cloud, 16, 8, WHITE);
    }
  }

  // Draw Obstacles
  for (int i = 0; i < MAX_OBSTACLES; i++) {
    if (obstacles[i].active) {
      if (obstacles[i].type == 0) {
        display.drawBitmap(obstacles[i].x, obstacles[i].y, cactus_small, 8, 16, WHITE);
      } else {
        display.drawBitmap(obstacles[i].x, obstacles[i].y, cactus_large, 16, 16, WHITE);
      }
    }
  }

  // Draw Dinosaur
  if (currentState == STATE_GAME_OVER) {
    // When dead, display the static frame
    display.drawBitmap(dino.x, dino.y, dino_jump, 16, 16, WHITE);
  } else if (dino.isJumping) {
    display.drawBitmap(dino.x, dino.y, dino_jump, 16, 16, WHITE);
  } else {
    if (dino.animationFrame == 0) {
      display.drawBitmap(dino.x, dino.y, dino_run1, 16, 16, WHITE);
    } else {
      display.drawBitmap(dino.x, dino.y, dino_run2, 16, 16, WHITE);
    }
  }

  // Draw Score
  display.setTextSize(1);
  display.setTextColor(WHITE);
  
  char scoreStr[16];
  sprintf(scoreStr, "HI:%04lu %04lu", highScore, score);
  // Each char is 6 pixels wide in size 1. (Length * 6 pixels)
  display.setCursor(SCREEN_WIDTH - 72 - 2, 0); 
  display.print(scoreStr);

  if (updateDisplay) {
    display.display();
  }
}

void showStartMenu() {
  display.clearDisplay();
  
  // Draw some scenery
  display.drawLine(0, 52, SCREEN_WIDTH, 52, WHITE);
  display.drawBitmap(80, 52 - 16, cactus_large, 16, 16, WHITE);
  display.drawBitmap(100, 52 - 16, cactus_small, 8, 16, WHITE);
  display.drawBitmap(20, 10, cloud, 16, 8, WHITE);
  display.drawBitmap(90, 15, cloud, 16, 8, WHITE);
  
  // Draw Dino running animation in place
  int dinoY = 52 - 16;
  if ((millis() / 200) % 2 == 0) {
    display.drawBitmap(14, dinoY, dino_run1, 16, 16, WHITE);
  } else {
    display.drawBitmap(14, dinoY, dino_run2, 16, 16, WHITE);
  }

  // Draw Title Popup
  display.fillRect(16, 8, 96, 32, BLACK);
  display.drawRect(16, 8, 96, 32, WHITE);
  
  display.setTextSize(1);
  display.setTextColor(WHITE);
  display.setCursor(37, 12);
  display.print("DINO GAME");

  // Flash text every 500ms
  if ((millis() / 500) % 2 == 0) {
    display.setCursor(22, 26);
    display.print("Press to Start");
  }

  display.display();
}

void showGameOver() {
  // Draw the exact game state where the player died
  drawGame(false);
  
  // Draw "GAME OVER" popup box
  int boxW = 104;
  int boxH = 34;
  int boxX = (SCREEN_WIDTH - boxW) / 2;
  int boxY = 8;
  
  display.fillRect(boxX, boxY, boxW, boxH, BLACK);
  display.drawRect(boxX, boxY, boxW, boxH, WHITE);

  display.setTextSize(1);
  display.setTextColor(WHITE);
  display.setCursor(boxX + 25, boxY + 5);
  display.print("GAME OVER");

  char scoreMsg[20];
  sprintf(scoreMsg, "Score: %lu", score);
  int scoreWidth = strlen(scoreMsg) * 6;
  display.setCursor(boxX + (boxW - scoreWidth) / 2, boxY + 15);
  display.print(scoreMsg);

  // Flash restart text
  if ((millis() / 500) % 2 == 0) {
    display.setCursor(boxX + 4, boxY + 25);
    display.print("Press to Restart");
  }

  display.display();
}
